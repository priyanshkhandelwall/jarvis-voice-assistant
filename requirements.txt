# JARVIS requirements. Pinned versions (2.4), including every transitive
# dependency. Tested in a clean Python 3.11 virtual environment on
# Windows 11 -- see README.md for run steps.
#
# Missing-setup rule (R14): every package below is optional except flet,
# and a missing one turns off only the feature that needs it. That
# feature's message says what is missing and what to do, e.g. "psutil
# isn't installed. Run pip install -r requirements.txt in the JARVIS
# folder, then restart JARVIS." Without flet there is no dashboard window;
# main.py then says: "The dashboard (F6) needs the 'flet' package, which
# is not installed. Install it with: pip install -r requirements.txt".
#
# F2 voice input and her Kokoro voice also need model files that pip can't
# install: run venv\Scripts\python.exe scripts\download_models.py once
# (README).

# ---- Direct dependencies --------------------------------------------------

# Test suite only; JARVIS itself runs without it.
pytest==9.1.1
# F6 dashboard. flet-desktop is the native window runtime: flet installs
# it with pip on first launch if it is missing, so it is pinned here to
# keep that install out of run time and at a known version.
flet==1.0.0
flet-desktop==1.0.0
# F6 gauges (CPU/RAM/disk/network) and R12 launch verification fallback.
psutil==7.2.2
# F3: real window-title verification (R12) for opening apps. Optional --
# jarvis/actions/open_app.py falls back to a coarser process-count check
# if this isn't installed (R14).
pywin32==312 ; sys_platform == "win32"
# F2's Q&A/small-talk piece (jarvis/speech/qa.py): Groq's free-tier API.
# A missing `groq` package or GROQ_API_KEY degrades to a spoken message
# saying exactly what to set up and where (decision C, R14), never a crash.
groq==1.7.0
python-dotenv==1.2.3
# Imported directly by tests/test_qa.py (also a groq and flet dependency).

# F2 her voice (jarvis/speech/neural_tts.py): Kokoro
# af_bella. kokoro-onnx (MIT) runs the Kokoro-82M model (Apache-2.0) on the
# CPU. Without it, or without its model, she speaks with Zira (below) and
# says exactly what to run (R14).
kokoro-onnx==0.6.1
# F2 fallback voice (jarvis/speech/tts.py): pyttsx3 drives Windows' SAPI
# voices (Microsoft Zira Desktop), used only when Kokoro isn't set up.
# Without it either, replies show as text and the dashboard says what to
# install (R14). comtypes and pypiwin32 are pyttsx3's own Windows
# dependencies (pypiwin32 is a shim over pywin32).
pyttsx3==2.99

# F2 Stage 2 voice input (jarvis/voice/): microphone capture (MIT; the

# without their models, typed commands still work and the dashboard says
# exactly what to set up (R14).

# F16b: searches open Google in the developer's own Chrome
